		if (amount>0)
			balance +=amount;
		else
			System.out.printf("Betrag muss positiv sein! Betrag: %.2f\n\n",amount);

		if(balance<amount && amount>0) {
			System.out.printf("Ihr Guthaben reicht nicht aus fr %.2f .\n"

			balance-=amount;
		else
			System.out.printf("Betrag muss positiv sein! Betrag: %.2f \n\n",amount);
	}

		if (this == obj)
			return true;

			return false;

				&& Objects.equals(creationDate, other.creationDate) && Objects.equals(owner, other.owner);
	}

		return "Account [accountID=" + accountID + ", balance=" + balance + ", creationDate=" + creationDate

        counter = 0;

package OOP;

import java.util.Arrays;

public class Matrix {

	private double[][] data;

	public Matrix(int zeilen, int spalten) {
		if (zeilen <= 0 || spalten <= 0) {
			throw new MatrixParseException("Ungltige Matrixgre");
		}
		data = new double[zeilen][spalten];
	}

	public Matrix(double[][] data) {

		if (data == null || data.length == 0 || data[0].length == 0) {
			throw new MatrixParseException("Matrix darf nicht leer sein.");
		}

		int cols = data[0].length;

		for (int i = 1; i < data.length; i++) {
			if (data[i].length != cols) {
				throw new MatrixParseException("Ungltige Matrixform.");
			}
		}

		this.data = new double[data.length][cols];

		for (int i = 0; i < data.length; i++) {
			System.arraycopy(data[i], 0, this.data[i], 0, cols);
		}
	}

	public int zeilen() {
		return data.length;
	}

	public int spalten() {
		return data[0].length;
	}

	public double get(int i, int j) {
		return data[i][j];
	}

	public void set(int i, int j, double value) {
		data[i][j] = value;
	}

	public Matrix add(Matrix other) {

		if (zeilen() != other.zeilen() || spalten() != other.spalten()) {
			throw new MatrixDimensionException("Dimension mismatch");
		}

		Matrix result = new Matrix(zeilen(), spalten());

		for (int i = 0; i < zeilen(); i++) {
			for (int j = 0; j < spalten(); j++) {
				result.set(i, j, this.get(i, j) + other.get(i, j));
			}
		}

		return result;
	}

	public Matrix subtract(Matrix other) {

		if (zeilen() != other.zeilen() || spalten() != other.spalten()) {
			throw new MatrixDimensionException("Dimension mismatch");
		}

		Matrix result = new Matrix(zeilen(), spalten());

		for (int i = 0; i < zeilen(); i++) {
			for (int j = 0; j < spalten(); j++) {
				result.set(i, j, this.get(i, j) - other.get(i, j));
			}
		}

		return result;
	}

	public Matrix multiply(Matrix other) {

		if (this.spalten() != other.zeilen()) {
			throw new MatrixDimensionException("Dimension mismatch");
		}

		Matrix result = new Matrix(this.zeilen(), other.spalten());

		for (int i = 0; i < this.zeilen(); i++) {
			for (int j = 0; j < other.spalten(); j++) {

				double sum = 0;

				for (int k = 0; k < this.spalten(); k++) {
					sum += this.get(i, k) * other.get(k, j);
				}

				result.set(i, j, sum);
			}
		}

		return result;
	}

	public double trace() {
		if (zeilen() != spalten()) {
			throw new MatrixDimensionException("Matrix muss quadratisch sein");
		}
		double spur = 0;
		for (int i = 0; i < zeilen(); i++) {
			spur += this.get(i, i);
		}
		return spur;
	}

	public Matrix transpose() {
		Matrix transponierte = new Matrix(spalten(), zeilen());
		for (int i = 0; i < spalten(); i++) {
			for (int j = 0; j < zeilen(); j++) {
				transponierte.set(i, j, this.get(j, i));
			}
		}
		return transponierte;
	}

	public Matrix scalarMultiply(double scalar) {
		Matrix result = new Matrix(zeilen(), spalten());
		for (int i = 0; i < zeilen(); i++) {
			for (int j = 0; j < spalten(); j++) {
				result.set(i, j, scalar * this.get(i, j));
			}
		}
		return result;
	}

	public double determinant() {
		if (this.zeilen() != this.spalten()) {
			throw new MatrixDimensionException("Die Determinante ist nur fuer quadratische Matrizen definiert.");
		}

		if (this.zeilen() == 1) {
			return this.get(0, 0);
		}

		if (this.zeilen() == 2) {
			return this.get(0, 0) * this.get(1, 1) - this.get(0, 1) * this.get(1, 0);
		}

		double summe = 0.0;
		for (int j = 0; j < this.spalten(); j++) {
			double vorzeichen = sign(0, j);
			summe += vorzeichen * this.get(0, j) * subMatrix(0, j).determinant();
		}
		return summe;
	}

	public Matrix inverse() {

		if (zeilen() != spalten()) {
			throw new MatrixDimensionException("Die Inverse ist nur fuer quadratische Matrizen definiert.");
		}

		double det = determinant();

		if (det == 0) {
			throw new MatrixNotInvertibleException("Die Matrix ist nicht invertierbar, da die Determinante 0 ist.");
		}

		if (zeilen() == 1) {
			Matrix result = new Matrix(1, 1);
			result.set(0, 0, 1.0 / get(0, 0));
			return result;
		}

		return adjugate().scalarMultiply(1.0 / det);
	}

	private Matrix subMatrix(int zeile, int spalte) {
		Matrix result = new Matrix(this.zeilen() - 1, this.spalten() - 1);

		int neueZeile = 0;
		for (int i = 0; i < this.zeilen(); i++) {
			if (i == zeile) {
				continue;
			}

			int neueSpalte = 0;
			for (int j = 0; j < this.spalten(); j++) {
				if (j == spalte) {
					continue;
				}

				result.set(neueZeile, neueSpalte, this.get(i, j));
				neueSpalte++;
			}
			neueZeile++;
		}

		return result;

	}

	private double sign(int row, int col) {
		return ((row + col) % 2 == 0) ? 1.0 : -1.0;
	}

	private Matrix adjugate() {
		int n = this.zeilen();
		Matrix kofaktoren = new Matrix(n, n);

		for (int i = 0; i < n; i++) {
			for (int j = 0; j < n; j++) {
				Matrix minor = subMatrix(i, j);
				double vorzeichen = sign(i, j);
				kofaktoren.set(i, j, vorzeichen * minor.determinant());
			}
		}

		return kofaktoren.transpose();
	}

	public String toString() {

		StringBuilder sb = new StringBuilder();

		for (int i = 0; i < zeilen(); i++) {
			for (int j = 0; j < spalten(); j++) {
				sb.append(String.format("%10.2f", this.get(i, j)));
			}
			sb.append("\n");
		}
		return sb.toString();
	}

	public int hashCode() {
		final int prime = 31;
		int result = 1;
		result = prime * result + Arrays.deepHashCode(data);
		return result;
	}

		if (this == obj)
			return true;
		if (obj == null)
			return false;
		if (getClass() != obj.getClass())

		Matrix other = (Matrix) obj;

	public static double berechneDeterminante(double[][] mat) {
	    int n = mat.length;

	    // Plausibilittsprfung: Die Matrix muss quadratisch sein
	    if (n != mat[0].length) {
	        throw new IllegalArgumentException("Die Matrix muss quadratisch sein.");
	    }

	    // --- BASISFALL ---
	    // Wenn wir bei einer 2x2 Matrix angekommen sind, nutzen wir deine Logik
	    if (n == 2) {
	        return mat[0][0] * mat[1][1] - mat[0][1] * mat[1][0];
	    }
	    
	    // (Optionaler Zusatz: Falls die Matrix 1x1 ist)
	    if (n == 1) {
	        return mat[0][0];
	    }

	    // --- REKURSIONSSCHRITT ---
	    double determinante = 0;

	    // Wir entwickeln nach der ersten Zeile (Index 0)
	    for (int j = 0; j < n; j++) {
	        // 1. Untermatrix erstellen (Gre (n-1) x (n-1))
	        double[][] untermatrix = erzeugeUntermatrix(mat, 0, j);

	        // 2. Das Vorzeichen bestimmen (Schachbrettmuster: + - + - ...)
	        // Fr die erste Zeile (i=0) gilt: (-1)^(0+j) -> also positiv bei geradem j, negativ bei ungeradem j
	        double vorzeichen = (j % 2 == 0) ? 1.0 : -1.0;

	        // 3. Rekursiver Aufruf und Aufsummieren
	        determinante += vorzeichen * mat[0][j] * berechneDeterminante(untermatrix);
	    }

	    return determinante;
	}

	// Hilfsmethode, die eine Zeile und eine Spalte aus der Matrix streicht
	private static double[][] erzeugeUntermatrix(double[][] mat, int zeileZuStreichen, int spalteZuStreichen) {
	    int n = mat.length;
	    double[][] untermatrix = new double[n - 1][n - 1];
	    
	    int r = 0; // Zeilen-Index fr die Untermatrix
	    for (int i = 0; i < n; i++) {
	        if (i == zeileZuStreichen) continue; // berspringe die Streichzeile
	        
	        int c = 0; // Spalten-Index fr die Untermatrix
	        for (int j = 0; j < n; j++) {
	            if (j == spalteZuStreichen) continue; // berspringe die Streichspalte
	            
	            untermatrix[r][c] = mat[i][j];
	            c++;
	        }
	        r++;
	    }
	    return untermatrix;
	}

	public static double[][] berechneInverse(double[][] mat) throws KeineInverseException{
		if(mat.length != 2 || mat[0].length != 2) {
			throw new UnsupportedOperationException(
				    "Inverse aktuell nur fr 2x2");
		}
		double[][] inverse= new double[mat.length][mat[0].length];
		if (berechneDeterminante(mat)==0) {
			throw new KeineInverseException();
		}
		double det=berechneDeterminante(mat);
		inverse[0][0]= mat[1][1]/det;
		inverse[0][1]= -mat[0][1]/det;
		inverse[1][0]= -mat[1][0]/det;
		inverse[1][1]=mat[0][0]/det;		
		return inverse;
	}

		int[][] transponierte = new int[a[0].length][a.length];
		for (int i = 0; i < a[0].length; i++) {
			for (int j = 0; j < a.length; j++) {
				transponierte[i][j] = a[j][i];
			}
		}
